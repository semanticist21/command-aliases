alias cl='claude --dangerously-skip-permissions --chrome'
alias cdx='codex --yolo'
function ccl() {
  local node_opts="${NODE_OPTIONS:-}"
  if [[ "$node_opts" != *"--max-old-space-size="* ]]; then
    node_opts="${node_opts:+$node_opts }--max-old-space-size=8192"
  fi
  NODE_OPTIONS="$node_opts" openclaude --dangerously-skip-permissions --chrome "$@"
}
